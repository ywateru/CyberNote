    1 import type { Editor } from "@tiptap/core";
    2 
    3 // Every existing editable restoration passes through this gate. A late access,
    4 // conflict-copy or paper callback can update a parked editor's desired state,
    5 // but cannot make it writable before that same instance becomes active again.
    6 // Callers keep their existing destroyed-instance checks; the gate only owns
    7 // editability and never changes editor lifetime or authorization.
    8 export class SearchEditorGate {
    9   private parked = new WeakMap<Editor, { editable: boolean }>();
   10 
   11   park(editor: Editor): void {
   12     if (!this.parked.has(editor))
   13       this.parked.set(editor, { editable: editor.isEditable });
   14     if (editor.isEditable) editor.setEditable(false, false);
   15   }
   16 
   17   activate(editor: Editor, blocked: boolean): void {
   18     if (blocked) {
   19       this.park(editor);
   20       return;
   21     }
   22     const desired = this.parked.get(editor);
   23     if (!desired) return;
   24     this.parked.delete(editor);
   25     if (editor.isEditable !== desired.editable)
   26       editor.setEditable(desired.editable, false);
   27   }
   28 
   29   restore(editor: Editor, editable: boolean, blocked: boolean): void {
   30     const desired = this.parked.get(editor);
   31     if (desired) desired.editable = editable;
   32     const allowed = editable && !desired && !blocked;
   33     if (editor.isEditable !== allowed) editor.setEditable(allowed, false);
   34   }
   35 }

    1 // Search selection is a projection. The real folder/note selection remains
    2 // untouched so clearing the query restores it, including its mounted editor.
    3 export function projectSearchNote(input: {
    4   searching: boolean;
    5   baseNoteId: string | null;
    6   searchNoteId: string | null;
    7   resultIds: readonly string[];
    8 }): string | null {
    9   if (!input.searching) return input.baseNoteId;
   10   if (input.resultIds.length === 1) return input.resultIds[0];
   11   const selected = input.searchNoteId || input.baseNoteId;
   12   return selected && input.resultIds.includes(selected) ? selected : null;
   13 }

   15 // Clearing an unrelated parked note must not silently change the visible
