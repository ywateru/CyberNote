// Reviewer-only overlay in a separate temporary crate snapshot; never author source.
#[tokio::test]
async fn reviewer_server_response_does_not_equate_all_errors_with_invalid_grant() {
    let cases = [
        (400, "invalid_grant", AuthError::RefreshFailed),
        (503, "server_error", AuthError::Network),
        (503, "temporarily_unavailable", AuthError::Network),
        (401, "invalid_client", AuthError::Configuration),
        (400, "reviewer_unknown_error", AuthError::TokenResponseInvalid),
    ];
    let mut observed = Vec::new();
    let mut expected = Vec::new();
    for (status, code, wanted) in cases {
        let f = Fixture::new();
        let dir = private_temp();
        let fault = Arc::new(AtomicUsize::new(0));
        let attempts = Arc::new(AtomicUsize::new(0));
        let mut http = f.http(&config());
        let fixture = f.clone();
        let failure = fault.clone();
        let requests = attempts.clone();
        http.responder = Some(Arc::new(move |request: HttpRequest| {
            let f = fixture.clone();
            let failure = failure.clone();
            let requests = requests.clone();
            Box::pin(async move {
                let refresh = request.uri().path() == "/oidc/token"
                    && url::form_urlencoded::parse(request.body())
                        .any(|(key, value)| key == "grant_type" && value == "refresh_token");
                if refresh {
                    requests.fetch_add(1, Ordering::SeqCst);
                    if failure.load(Ordering::SeqCst) == 1 {
                        return Ok(response(status, json!({"error":code,"error_description":"synthetic-sensitive-text"})));
                    }
                }
                f.respond(request).await
            })
        }));
        let path = dir.path().join("grant.json");
        let client = AuthClient::initialize(config(), &path, http.clone()).unwrap();
        f.login(&client).await;
        drop(client);
        let before = fs::read(&path).unwrap();
        let client = AuthClient::initialize(config(), &path, http).unwrap();
        fault.store(1, Ordering::SeqCst);
        let error = client.restore().await.unwrap_err();
        assert!(!error.to_string().contains("synthetic-sensitive"));
        assert_eq!(attempts.load(Ordering::SeqCst), 1);
        assert_eq!(fs::read(&path).unwrap(), before);
        assert!(!client.cleanup_required().await);
        assert_eq!(f.revokes.load(Ordering::SeqCst), 0);
        println!("review-case http={status} kind={code} observed={error:?}");
        observed.push(error);
        expected.push(wanted);
    }
    // Fails at 7e8ddd17: all five cases collapse to RefreshFailed.
    assert_eq!(observed, expected, "Only invalid_grant confirms grant rejection; temporary/config/unknown errors must remain distinct");
}
