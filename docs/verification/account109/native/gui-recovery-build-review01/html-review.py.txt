class Canonical(HTMLParser):
    void = {'area', 'base', 'br', 'col', 'embed', 'hr', 'img', 'input', 'link', 'meta', 'param', 'source', 'track', 'wbr'}
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.events = []
        self.refs = []
    def handle_decl(self, data):
        self.events.append(('declaration', data.lower()))
    def handle_starttag(self, tag, attrs):
        normalized = sorted((k, '' if v is None else v) for k, v in attrs)
        self.events.append(('start', tag, normalized))
        if tag in {'script', 'link'}:
            self.refs.extend(v for k, v in attrs if k in {'src', 'href'})
    def handle_startendtag(self, tag, attrs):
        self.handle_starttag(tag, attrs)
        if tag not in self.void:
            self.handle_endtag(tag)
    def handle_endtag(self, tag):
        if tag not in self.void:
            self.events.append(('end', tag))
    def handle_data(self, data):
        if data.strip():
            self.events.append(('data', data))
    def handle_comment(self, data):
        self.events.append(('comment', data))

a, b = Canonical(), Canonical()
a.feed(dist.decode()); a.close()
b.feed(decoded.decode()); b.close()
assert a.events == b.events, 'HTML structural comparison failed'
assert a.refs == b.refs
report = json.loads((out / 'review-result.json').read_text())

    normalization='HTMLParser event comparison: declaration case, attribute order, valueless vs empty attributes, self-closing tag spelling, and whitespace-only text nodes normalized; non-whitespace text and script contents retained exactly.',
    limitation='This bounded structural comparison is not full browser DOM/runtime equivalence and does not claim raw index.html byte equality.')
