781: /* Look for a named file inside the AVD's content directory.
782:  * Returns NULL if it doesn't exist, or a strdup() copy otherwise.
783:  */
784: static char* _avdInfo_getContentFilePath(const AvdInfo* i,
785:                                          const char* fileName) {
786:     char temp[MAX_PATH], *p = temp, *end = p + sizeof(temp);
787: 
788:     p = bufprint(p, end, "%s" PATH_SEP "%s", i->contentPath, fileName);
789:     if (p >= end) {
790:         derror("can't access virtual device content directory");
791:         return NULL;
792:     }
793:     if (!path_exists(temp)) {
794:         return NULL;
795:     }
796:     return ASTRDUP(temp);
797: }

1536: static char* get_device_path(const AvdInfo* info, const char* image) {
1537:     const char* device_table[6] = {"", "", "", "", "", ""};
1538:     int i = 0;
1539:     device_table[i++] = "sdcard";
1540:     if (has_encryption(info)) {
1541:         device_table[i++] = "encryption";
1542:     }
1543:     device_table[i++] = "system";
1544:     device_table[i++] = "userdata";
1545:     if (has_vendor(info)) {
1546:         device_table[i++] = "vendor";
1547:     }
1548:     device_table[i++] = "cache";
1549:     int count = ARRAY_SIZE(device_table);
1550:     for (i = 0; i < count; ++i) {
1551:         if (strcmp(image, device_table[i]) == 0) {
1552:             break;
1553:         }
1554:     }
1555:     if (i == count) {
1556:         return NULL;
1557:     }
1558:     char buf[1024];
1559: 
1560:     if (is_armish(info)) {
1561:         snprintf(buf, sizeof(buf),
1562:                  "/dev/block/platform/%s.virtio_mmio/by-name/%s",
1563:                  arm_device_id[i], image);
1564:     }
1565:     return strdup(buf);
1566: }
