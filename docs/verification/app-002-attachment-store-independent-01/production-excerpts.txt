  270     pub(crate) fn query_attachment_operation(
  271         &self,
  272         operation: &str,
  273         note: &str,
  274         owner: &AttachmentOwner,
  275     ) -> Result<Value> {
  276         let Some(op) = self.owned_operation(operation, note, owner)? else {
  277             return Ok(json!({"operationId":operation,"noteId":note,"status":"notPersisted"}));
  278         };
  279         if matches!(op.state.as_str(), "aborted" | "retired") {
  280             return response(&op, "aborted", op.reason.as_deref());
  281         }
  282         match self.operation_reference(&op)? {
  283             Reference::Present => {
  284                 self.check_operation_file(&op)?;
  285                 self.set_operation_state(&op, "committed", None)?;
  286                 response(&op, "committed", None)
  287             }
  288             Reference::Uncertain(reason) => response(&op, "deferred", Some(reason)),
  289             Reference::Absent if op.state == "stored" || op.state == "committed" => {
  290                 self.check_operation_file(&op)?;
  291                 response(&op, &op.state, None)
  292             }
  293             Reference::Absent => response(
  294                 &op,
  295                 "deferred",
  296                 Some(if op.state == "aborting" {
  297                     "cleanup_pending"
  298                 } else {
  299                     "storage_incomplete"
  300                 }),
  301             ),
  302         }
  303     }
  304     pub(crate) fn defer_attachment_operation(
  305         &self,
  306         operation: &str,
  307         note: &str,
  308         owner: &AttachmentOwner,
  309         reason: &str,
  310     ) -> Result<Value> {
  311         let value = self.query_attachment_operation(operation, note, owner)?;
  312         if value["status"] == "committed"
  313             || value["status"] == "aborted"
  314             || value["status"] == "notPersisted"
  315         {
  316             return Ok(value);
  317         }
  318         Ok(json!({"operationId":operation,"noteId":note,"status":"deferred","reason":reason}))
  319     }
  320     pub(crate) fn cancel_attachment_operation(
  321         &mut self,
  322         operation: &str,
  323         note: &str,
  324         owner: &AttachmentOwner,
  325     ) -> Result<Value> {
  326         let Some(op) = self.owned_operation(operation, note, owner)? else {
  327             return Ok(json!({"operationId":operation,"noteId":note,"status":"notPersisted"}));
  328         };
  329         if matches!(op.state.as_str(), "aborted" | "retired") {
  330             return response(&op, "aborted", op.reason.as_deref());
  331         }
  332         self.cancel_owned_attachment_operation(&op)
  333     }
  334     fn cancel_owned_attachment_operation(&mut self, op: &Operation) -> Result<Value> {
  335         match self.operation_reference(op)? {
  336             Reference::Present => {
  337                 self.check_operation_file(op)?;
  338                 self.set_operation_state(op, "committed", None)?;
  339                 return response(op, "committed", None);
  340             }
  341             Reference::Uncertain(reason) => return response(op, "deferred", Some(reason)),
  342             Reference::Absent => {}
  343         }
  344         if op.state == "committed" {
  345             return response(op, "committed", None);
  346         }
  347         // Withdraw metadata first: a later newly registered save cannot refer
  348         // to an attachment whose cancellation has already begun.
  349         let tx = self.db.transaction()?;
  350         tx.execute(
  351             "DELETE FROM attachments WHERE id=?1 AND note_id=?2",
  352             params![op.attachment.id, op.note],
  353         )?;
  354         tx.execute("UPDATE attachment_operations SET state='aborting',reason='cleanup_pending' WHERE operation_id=?1",[&op.id])?;
  355         tx.commit()?;
  356         checkpoint("after_cancel_metadata");
  357         match fs::remove_file(self.attachment_path(&op.attachment.relative_path)?) {
  358             Ok(()) => {}
  359             Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
  360             Err(error) => return Err(error.into()),
  361         }
  362         sync_directory(&self.root.join("attachments"))?;
  363         checkpoint("after_cancel_file");
  364         self.set_operation_state(op, "aborted", None)?;
  365         response(op, "aborted", None)
  366     }
  367     pub(super) fn operation_files_for_purge(&self, note: &str) -> Result<Vec<Attachment>> {
  368         let ids = {
  369             let mut query = self.db.prepare("SELECT operation_id FROM attachment_operations WHERE note_id=?1 AND state NOT IN ('aborted','retired')")?;
  370             let ids = query
  371                 .query_map([note], |row| row.get::<_, String>(0))?
  372                 .collect::<std::result::Result<Vec<_>, _>>()?;
  373             ids
  374         };
  375         let live = self.attachments()?;
  376         let mut files = Vec::new();
  377         for id in ids {
  378             let op = self
  379                 .attachment_operation(&id)?
  380                 .ok_or_else(|| err("attachment_operation_missing", "附件操作记录消失"))?;
  381             let attachment = op.attachment;
  382             if attachment.note_id != note
  383                 || attachment.relative_path != format!("op-{}-{}.blob", op.id, attachment.id)
  384                 || live.iter().any(|other| {
  385                     (other.id == attachment.id || other.relative_path == attachment.relative_path)
  386                         && other != &attachment
  387                 })
  388             {
  389                 return Err(conflict());
  390             }
  391             self.attachment_path(&attachment.relative_path)?;
  392             files.push(attachment);
  393         }
  394         Ok(files)
  395     }
  396     pub(super) fn recover_attachment_operations(&mut self) -> Result<()> {
  397         let ids = {
  398             let mut query=self.db.prepare("SELECT operation_id FROM attachment_operations WHERE state IN ('preparing','stored','aborting')")?;
  399             let ids = query
  400                 .query_map([], |row| row.get::<_, String>(0))?
  401                 .collect::<std::result::Result<Vec<_>, _>>()?;
  402             ids
  403         };
  404         for operation in ids {
  405             let op = self
  406                 .attachment_operation(&operation)?
  407                 .ok_or_else(|| err("attachment_operation_missing", "恢复时附件操作消失"))?;
  408             // Only this private journal's nonterminal creations are eligible.
  409             // Old attachments/orphans and terminal histories are never swept.
  410             self.cancel_owned_attachment_operation(&op)?;
  411         }
  412         Ok(())
  413     }
